\newlength{\reportDiagramHeight}
\newcommand{\fullPageAnalysisDiagram}[4][0]{%
	\clearpage
	\setlength{\reportDiagramHeight}{%
		\dimexpr\textheight-3\baselineskip\relax}
	\begin{figure}[p]
		\centering
		\adjustbox{max width=\textwidth,max height=\reportDiagramHeight}{%
			\rotatebox{#1}{\includegraphics{build/staruml/#2.pdf}}}
		\caption{#3}\label{#4}
	\end{figure}
	\clearpage
}
