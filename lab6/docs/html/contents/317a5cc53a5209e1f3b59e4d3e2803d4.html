Description

Задание анализа CHECK ((status IN ('completed','cancelled','failed')) = (finished_at IS NOT NULL)) CHECK (started_at IS NULL OR started_at >= created_at) CHECK (finished_at IS NULL OR finished_at >= COALESCE(started_at, created_at))
